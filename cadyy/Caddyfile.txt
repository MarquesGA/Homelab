# ------------- Jellyfin -----------------
jellyfin.example.com {
    tls {
        dns cloudflare {env.CF_API_TOKEN}  # ← Substitua pelo seu token Cloudflare
        propagation_delay 2m
        resolvers 1.1.1.1
    }

    reverse_proxy <JELLYFIN_IP>:8096
}

# ------------- Proxmox -----------------

    reverse_proxy https://<PROXMOX_IP>:8006 {
        transport http {
            tls
            tls_insecure_skip_verify
        }
    }
}
